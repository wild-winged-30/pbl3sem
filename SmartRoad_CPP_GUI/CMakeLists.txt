cmake_minimum_required(VERSION 3.16)
project(SmartRoadGUI LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
find_package(OpenCV REQUIRED)
add_executable(smartroad_gui src/main.cpp src/RoadReport.cpp src/ReportManager.cpp src/ImageManager.cpp)
target_include_directories(smartroad_gui PRIVATE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(smartroad_gui PRIVATE Qt5::Widgets ${OpenCV_LIBS})
